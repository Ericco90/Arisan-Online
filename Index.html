KODE BACKEND (Simpan sebagai Code.gs) // ========================================== function doGet() { return HtmlService.createHtmlOutputFromFile('Index') .setTitle('Arisan Online') .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } function getSheet() { return SpreadsheetApp.getActiveSpreadsheet(); } function setupDatabase() { const ss = getSheet(); // Sheet Anggota if (!ss.getSheetByName('Anggota')) { const sheet = ss.insertSheet('Anggota'); sheet.appendRow(['id', 'nama', 'telepon', 'sudah_bayar', 'pernah_menang']); sheet.appendRow(['M1001', 'Budi Santoso', '081234567890', true, false]); sheet.appendRow(['M1002', 'Siti Aminah', '081298765432', true, false]); sheet.appendRow(['M1003', 'Andi Hermawan', '085612341234', false, false]); sheet.appendRow(['M1004', 'Rina Marlina', '089988776655', true, false]); sheet.appendRow(['M1005', 'Dedi Kusuma', '087766554433', true, true]); sheet.autoResizeColumns(1, 5); } // Sheet Riwayat if (!ss.getSheetByName('Riwayat')) { const sheet = ss.insertSheet('Riwayat'); sheet.appendRow(['id', 'nama_pemenang', 'periode', 'jumlah', 'tanggal']); const dateStr = new Date().toISOString().split('T')[0]; sheet.appendRow(['H1001', 'Dedi Kusuma', "1-1", 500000, dateStr]); sheet.autoResizeColumns(1, 5); } // Sheet Pengaturan if (!ss.getSheetByName('Pengaturan')) { const sheet = ss.insertSheet('Pengaturan'); sheet.appendRow(['kunci', 'nilai']); sheet.appendRow(['pin_admin', '123456']); sheet.appendRow(['iuran', '100000']); sheet.appendRow(['periode', '1']); // Ini adalah Putaran/Bulan ke- sheet.appendRow(['siklus', '1']); // TAMBAHAN BARU: Ini adalah Periode Arisan ke- sheet.appendRow(['nama_bank', 'BCA / DANA']); sheet.appendRow(['no_rekening', '081234567890']); sheet.appendRow(['nama_pemilik', 'Budi Santoso (Admin)']); const nextMonth = new Date(); nextMonth.setMonth(nextMonth.getMonth() + 1); const nextDateStr = nextMonth.toISOString().split('T')[0]; sheet.appendRow(['tanggal_berikutnya', nextDateStr]); sheet.autoResizeColumns(1, 2); } // Hapus Sheet1 bawaan const defaultSheet = ss.getSheetByName('Sheet1'); if (defaultSheet && ss.getSheets().length > 1) { ss.deleteSheet(defaultSheet); } return true; } function getSheetData(sheetName) { const sheet = getSheet().getSheetByName(sheetName); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; const headers = data[0]; const rows = []; const dict = { 'id': 'id', 'nama': 'name', 'telepon': 'phone', 'sudah_bayar': 'isPaid', 'pernah_menang': 'hasWon', 'nama_pemenang': 'winnerName', 'periode': 'period', 'jumlah': 'amount', 'tanggal': 'date', 'kunci': 'key', 'nilai': 'value', 'pin_admin': 'adminPin', 'iuran': 'contribution', 'nama_bank': 'bankName', 'no_rekening': 'accountNumber', 'nama_pemilik': 'accountName', 'tanggal_berikutnya': 'nextDate', 'siklus': 'cycle' // TAMBAHAN BARU: Membaca Periode }; for (let i = 1; i < data.length; i++) { let obj = {}; for (let j = 0; j < headers.length; j++) { let header = headers[j]; let val = data[i][j]; let engKey = dict[header] || header; if (sheetName === 'Pengaturan' && header === 'kunci') { obj[engKey] = dict[val] || val; } else { obj[engKey] = val; } } if (obj.isPaid === 'TRUE' || obj.isPaid === true) obj.isPaid = true; else if (obj.isPaid === 'FALSE' || obj.isPaid === false) obj.isPaid = false; if (obj.hasWon === 'TRUE' || obj.hasWon === true) obj.hasWon = true; else if (obj.hasWon === 'FALSE' || obj.hasWon === false) obj.hasWon = false; rows.push(obj); } return rows; } function getAppData() { setupDatabase(); const members = getSheetData('Anggota'); const history = getSheetData('Riwayat'); const configRaw = getSheetData('Pengaturan'); let config = {}; configRaw.forEach(row => { config[row.key] = row.value; }); history.reverse(); return { members, history, config }; } function addMember(name, phone) { const sheet = getSheet().getSheetByName('Anggota'); const id = 'M' + new Date().getTime(); sheet.appendRow([id, name, phone, false, false]); return getAppData(); } // FUNGSI BARU: Edit data peserta function editMember(memberId, newName, newPhone) { const sheet = getSheet().getSheetByName('Anggota'); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] == memberId) { sheet.getRange(i + 1, 2).setValue(newName); sheet.getRange(i + 1, 3).setValue(newPhone); break; } } return getAppData(); } function togglePayment(id, currentStatus) { const sheet = getSheet().getSheetByName('Anggota'); const data = sheet.getDataRange().getValues(); const newStatus = !currentStatus; for (let i = 1; i < data.length; i++) { if (data[i][0] == id) { sheet.getRange(i + 1, 4).setValue(newStatus); break; } } return getAppData(); } function saveWinner(winnerId, winnerName, amount, period) { const ss = getSheet(); const memberSheet = ss.getSheetByName('Anggota'); const historySheet = ss.getSheetByName('Riwayat'); const configSheet = ss.getSheetByName('Pengaturan'); // 1. Tandai member menang const data = memberSheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] == winnerId) { memberSheet.getRange(i + 1, 5).setValue(true); break; } } // 2. Ambil Siklus & Putaran Saat Ini untuk dicatat di Riwayat const confData = configSheet.getDataRange().getValues(); let currentPeriod = 1; let currentSiklus = 1; let periodRow = -1; for (let i = 1; i < confData.length; i++) { if (confData[i][0] == 'periode') { currentPeriod = parseInt(confData[i][1]) || 1; periodRow = i + 1; } if (confData[i][0] == 'siklus') { currentSiklus = parseInt(confData[i][1]) || 1; } } // 3. Tambah ke Riwayat dengan format informatif (Misal: "1-2" -> Periode 1 Putaran 2) const histId = 'H' + new Date().getTime(); const dateStr = new Date().toISOString().split('T')[0]; const displayPeriod = currentSiklus + "-" + currentPeriod; historySheet.appendRow([histId, winnerName, displayPeriod, amount, dateStr]); // 4. OTOMATIS TAMBAHKAN ANGKA PUTARAN if (periodRow > -1) { configSheet.getRange(periodRow, 2).setValue(currentPeriod + 1); } return getAppData(); } // FUNGSI BARU: Hapus riwayat dan batalkan status pemenang function deleteHistory(historyId) { const ss = getSheet(); const historySheet = ss.getSheetByName('Riwayat'); const memberSheet = ss.getSheetByName('Anggota'); const configSheet = ss.getSheetByName('Pengaturan'); const hData = historySheet.getDataRange().getValues(); let winnerName = ""; // Cari ID riwayat dari bawah for (let i = hData.length - 1; i >= 1; i--) { if (hData[i][0] == historyId) { winnerName = hData[i][1]; historySheet.deleteRow(i + 1); break; } } // Jika ketemu namanya, kembalikan status pernah_menang = false (biar bisa diundi lagi) if (winnerName) { const mData = memberSheet.getDataRange().getValues(); for (let i = 1; i < mData.length; i++) { if (mData[i][1] == winnerName) { memberSheet.getRange(i + 1, 5).setValue(false); break; } } // Karena riwayat dihapus, angka putaran dikembalikan / dikurangi 1 const confData = configSheet.getDataRange().getValues(); for (let i = 1; i < confData.length; i++) { if (confData[i][0] == 'periode') { let cp = parseInt(confData[i][1]) || 1; if (cp > 1) { configSheet.getRange(i + 1, 2).setValue(cp - 1); } break; } } } return getAppData(); } function startNewPeriod() { const ss = getSheet(); const memberSheet = ss.getSheetByName('Anggota'); // Fungsi ini sekarang HANYA MENG-RESET STATUS PEMBAYARAN, // karena angka Putaran sudah otomatis naik saat mengocok (di fungsi saveWinner) const data = memberSheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { memberSheet.getRange(i + 1, 4).setValue(false); } return getAppData(); } function saveSettings(contribution, nextDate, adminPin, bankName, accountNumber, accountName) { const configSheet = getSheet().getSheetByName('Pengaturan'); const confData = configSheet.getDataRange().getValues(); for (let i = 1; i < confData.length; i++) { if (confData[i][0] == 'iuran') configSheet.getRange(i + 1, 2).setValue(contribution); if (confData[i][0] == 'tanggal_berikutnya') configSheet.getRange(i + 1, 2).setValue(nextDate); if (confData[i][0] == 'pin_admin') configSheet.getRange(i + 1, 2).setValue(adminPin); if (confData[i][0] == 'nama_bank') configSheet.getRange(i + 1, 2).setValue(bankName); if (confData[i][0] == 'no_rekening') configSheet.getRange(i + 1, 2).setValue(accountNumber); if (confData[i][0] == 'nama_pemilik') configSheet.getRange(i + 1, 2).setValue(accountName); } return getAppData(); } function resetCycle() { const ss = getSheet(); const memberSheet = ss.getSheetByName('Anggota'); const configSheet = ss.getSheetByName('Pengaturan'); // Reset semua peserta agar bisa diundi lagi const memberData = memberSheet.getDataRange().getValues(); for (let i = 1; i < memberData.length; i++) { memberSheet.getRange(i + 1, 4).setValue(false); memberSheet.getRange(i + 1, 5).setValue(false); } // Reset Putaran ke 1 & Naikkan Periode Arisan const confData = configSheet.getDataRange().getValues(); let hasSiklus = false; let currentSiklus = 1; for (let i = 1; i < confData.length; i++) { if (confData[i][0] == 'periode') { configSheet.getRange(i + 1, 2).setValue(1); // Kembali ke Putaran 1 } if (confData[i][0] == 'siklus') { hasSiklus = true; currentSiklus = parseInt(confData[i][1]) || 1; configSheet.getRange(i + 1, 2).setValue(currentSiklus + 1); // Lanjut Periode ke 2, 3, dst.. } } // Jika database lama belum punya kolom siklus, otomatis buatkan if (!hasSiklus) { configSheet.appendRow(['siklus', '2']); } return getAppData(); } // FUNGSI API PENTING: Menerima Perintah dari Luar (Website Hosting) function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) throw new Error("Data Kosong"); const payload = JSON.parse(e.postData.contents); const action = payload.action; const args = payload.args || []; let result; if (action === 'getAppData') result = getAppData(); else if (action === 'addMember') result = addMember(args[0], args[1]); else if (action === 'editMember') result = editMember(args[0], args[1], args[2]); else if (action === 'togglePayment') result = togglePayment(args[0], args[1]); else if (action === 'saveWinner') result = saveWinner(args[0], args[1], args[2], args[3]); else if (action === 'deleteHistory') result = deleteHistory(args[0]); else if (action === 'startNewPeriod') result = startNewPeriod(); else if (action === 'saveSettings') result = saveSettings(args[0], args[1], args[2], args[3], args[4], args[5]); else if (action === 'resetCycle') result = resetCycle(); return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: result })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
